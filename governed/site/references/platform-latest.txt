# --- ABSTRACT LMS ADAPTER INTERFACE ---
class BaseLMSAdapter(ABC):
    @abstractmethod
    def sync_user(self, user_data: Dict[str, Any]) -> bool:
        pass

    @abstractmethod
    def sync_course_grade(self, student_id: str, section_id: str, grade: str) -> bool:
        pass

# --- CUSTOM CANVAS LMS IMPLEMENTATION (REST + LTI 1.3) ---
class CanvasLMSAdapter(BaseLMSAdapter):
    def __init__(self, base_url: str, api_token: str):
        self.base_url = base_url.rstrip('/')
        self.headers = {
            "Authorization": f"Bearer {api_token}",
            "Content-Type": "application/json"
        }

    def sync_user(self, user_data: Dict[str, Any]) -> bool:
        """Provision user into Canvas LMS via REST API."""
        endpoint = f"{self.base_url}/api/v1/accounts/self/users"
        payload = {
            "user": {
                "name": f"{user_data['first_name']} {user_data['last_name']}",
                "short_name": user_data['first_name'],
                "sortable_name": f"{user_data['last_name']}, {user_data['first_name']}",
                "terms_of_use": True
            },
            "pseudonym": {
                "unique_id": user_data['email'],
                "send_confirmation": False
            }
        }
        res = requests.post(endpoint, json=payload, headers=self.headers)
        return res.status_code in (200, 201)

    def sync_course_grade(self, student_id: str, section_id: str, grade: str) -> bool:
        """Submit grade back to Canvas SpeedGrader via LTI 1.3 / REST API."""
        endpoint = f"{self.base_url}/api/v1/courses/{section_id}/assignments/grade"
        payload = {
            "grade_data": {
                student_id: {
                    "posted_grade": grade
                }
            }
        }
        res = requests.post(endpoint, json=payload, headers=self.headers)
        return res.status_code == 200

# --- QUAD-LMS FACTORY ---

                api_token=os.getenv("CANVAS_API_TOKEN", "mock_token")
            )
            # Adapters for Cloud-SaaS, On-Premise, and Open-Source instances mount here
        }

    def get_adapter(self, lms_type: str) -> BaseLMSAdapter:
        adapter = self.adapters.get(lms_type.lower())
        if not adapter:
            raise HTTPException(status_code=400, detail=f"Unsupported LMS type: {lms_type}")
        return adapter
